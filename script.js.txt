function hlHTML(s){
  return lex(s,/(<!--[\s\S]*?-->)|(<(style|script)\b[^>]*>)([\s\S]*?)(<\/\3>)|(<!DOCTYPE[^>]*>)|(<\/?)([\w:-]+)((?:"[^"]*"|'[^']*'|[^>"'])*)(\/?>)/gi,function(m){

  if(m[6])return sp('t-tag',m[6]);
  return sp('t-tag',m[7]+m[8])+lex(m[9],/("[^"]*"|'[^']*')|([\w:@.-]+)/g,function(a){return a[1]?sp('t-str',a[1]):sp('t-attr',a[2])})+sp('t-tag',m[10]);
});
}
function hlCSS(s){

  if(m[1])return sp('t-com',m[1]);if(m[2])return sp('t-str',m[2]);if(m[3])return sp('t-at',m[3]);

}

function fmtCSS(s){
  if(!s.trim())return '';
  var t=s.replace(/\s+/g,' ').replace(/\s*\{\s*/g,' {\n').replace(/;(?!base64|charset)\s*/g,';\n').replace(/\s*\}\s*/g,'\n}\n').replace(/\*\/\s*/g,'*/\n'),d=0;
  return t.split('\n').map(function(l){l=l.trim();if(!l)return null;if(l[0]==='}')d=Math.max(0,d-1);var o='  '.repeat(d)+l;if(/\{$/.test(l))d++;return o}).filter(function(x){return x!==null}).join('\n').replace(/\}\n(?=[^}\s])/g,'}\n\n');
}
function fmtJS(s){
  var d=0;
  return s.split('\n').map(function(l){
      l=l.trim();if(!l)return '';
      var t=l.replace(/"(?:\\.|[^"\\])*"|'(?:\\.|[^'\\])*'|`(?:\\.|[^`\\])*`|\/\/.*/g,'');
var o=(t.match(/[{[(]/g)||[]).length,c=(t.match(/[}\])]/g)||[]).length,lead=(t.match(/^[}\])]+/)||[''])[0].length;
var ind=Math.max(0,d-lead);d=Math.max(0,d+o-c);return '  '.repeat(ind)+l;
}).join('\n');
}
function fmtHTML(src){
  var tk=src.split(/(<!--[\s\S]*?-->|<(?:script|style)\b[\s\S]*?<\/(?:script|style)>|<[^>]+>)/i),o=[],d=0,I=function(){return '  '.repeat(d)};
  for(var i=0;i<tk.length;i++){
    var t=tk[i];
    if(i%2===0){t=t.replace(/\s+/g,' ').trim();if(t)o.push(I()+t);continue}
    if(/^<(script|style)\b/i.test(t)){
      var mm=/^(<[^>]*>)([\s\S]*?)(<\/[^>]*>)$/.exec(t),inner=/^<style/i.test(mm[1])?fmtCSS(mm[2]):fmtJS(mm[2].trim());
      if(!inner.trim()){o.push(I()+mm[1]+mm[3]);continue}
      o.push(I()+mm[1]);d++;inner.split('\n').forEach(function(l){o.push(l?I()+l:'')});d--;o.push(I()+mm[3]);continue;
    }
    var m=/^<(\/?)([\w!:-]+)/.exec(t);
    if(!m||/^<!--/.test(t)){o.push(I()+t);continue}
    var n=m[2].toLowerCase();
    if(m[1]){d=Math.max(0,d-1);o.push(I()+t);continue}
    if(VOID.test(n)||/\/>$/.test(t)){o.push(I()+t);continue}
    var nx=(tk[i+1]||'').replace(/\s+/g,' ').trim(),cl=(tk[i+2]||'').toLowerCase();
    if(nx&&nx.length<80&&cl==='</'+n+'>'){o.push(I()+t+nx+tk[i+2]);i+=2;continue}
    o.push(I()+t);d++;
  }
  return o.join('\n');
}
function fmtBy(k,v){return k==='css'?fmtCSS(v):k==='js'?fmtJS(v):fmtHTML(v)}

function parts(src){
  var d=new DOMParser().parseFromString(src,'text/html'),css=[],js=[];
  d.querySelectorAll('style').forEach(function(e){css.push(e.textContent.trim());e.remove()});
  d.querySelectorAll('script:not([src])').forEach(function(e){js.push(e.textContent.trim());e.remove()});
  var doc=/<html|<!doctype|<head|<body/i.test(src);
  return{html:doc?'<!DOCTYPE html>\n'+d.documentElement.outerHTML:d.body.innerHTML,css:css.join('\n\n'),js:js.join('\n\n')};
}

function chkHTML(v){
  var s=v.replace(/<!--[\s\S]*?-->/g,'').replace(/<(script|style)\b[\s\S]*?<\/\1>/gi,'<$1></$1>');
  var re=/<(\/?)([a-zA-Z][\w:-]*)[^>]*?(\/?)>/g,m,st=[],OPT=/^(p|li|td|th|tr|thead|tbody|tfoot|option|dt|dd|html|body|head)$/i,req=function(x){return!OPT.test(x)};
  while((m=re.exec(s))){
    var n=m[2].toLowerCase();if(VOID.test(n)||m[3])continue;
    if(!m[1]){st.push(n);continue}
    var i=st.lastIndexOf(n);
    if(i<0){if(req(n))return 'Unexpected closing </'+n+'>';continue}
    var bad=st.slice(i+1).filter(req);if(bad.length)return '<'+bad[bad.length-1]+'> is never closed';
    st.length=i;
  }
  var left=st.filter(req);return left.length?'<'+left[left.length-1]+'> is never closed':null;
}
